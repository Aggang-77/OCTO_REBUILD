cmake_minimum_required(VERSION 3.25)
project(octo_rebuild VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OCTO_ASAN "Enable AddressSanitizer" OFF)
add_library(octo_physics core/world.cpp)
target_include_directories(octo_physics PUBLIC include)
if(MSVC)
 target_compile_options(octo_physics PRIVATE /W4 /permissive- /utf-8 /fp:precise)
 if(OCTO_ASAN)
  target_compile_options(octo_physics PUBLIC /fsanitize=address /Z7)
  target_link_options(octo_physics PUBLIC /INCREMENTAL:NO)
 endif()
else()
 target_compile_options(octo_physics PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(octo_contact_mantle_tests tests/contact_mantle_tests.cpp)
target_link_libraries(octo_contact_mantle_tests PRIVATE octo_physics)
add_test(NAME physics.contact_mantle COMMAND octo_contact_mantle_tests)
add_executable(octo_volumetric_arm_tests tests/volumetric_arm_tests.cpp)
target_link_libraries(octo_volumetric_arm_tests PRIVATE octo_physics)
add_test(NAME physics.volumetric_arm COMMAND octo_volumetric_arm_tests)
add_executable(octo_viscoelastic_tests tests/viscoelastic_tests.cpp)
target_link_libraries(octo_viscoelastic_tests PRIVATE octo_physics)
add_test(NAME physics.viscoelastic COMMAND octo_viscoelastic_tests)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/controls_v1_tests.cpp")
 add_executable(octo_controls_v1_tests tests/controls_v1_tests.cpp)
 target_link_libraries(octo_controls_v1_tests PRIVATE octo_physics)
 add_test(NAME controls.v1 COMMAND octo_controls_v1_tests)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/physics_tests.cpp")
 add_executable(octo_physics_tests tests/physics_tests.cpp)
 target_link_libraries(octo_physics_tests PRIVATE octo_physics)
 add_test(NAME physics.unit COMMAND octo_physics_tests)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/bridge/server.cpp")
 add_executable(octo_server bridge/server.cpp)
 target_link_libraries(octo_server PRIVATE octo_physics)
 if(WIN32 AND EXISTS "C:/Anaconda/python.exe" AND NOT Python3_EXECUTABLE)
  set(Python3_EXECUTABLE "C:/Anaconda/python.exe" CACHE FILEPATH "Local Python interpreter")
 endif()
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND)
  add_test(NAME bridge.protocol COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/bridge_protocol_tests.py" $<TARGET_FILE:octo_server>)
  find_program(OCTO_GODOT_EXECUTABLE NAMES godot godot.exe PATHS "C:/Tools/Godot")
  if(OCTO_GODOT_EXECUTABLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/game/assets/Octopus_GameReady_v5.glb")
   add_test(NAME game.input COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/godot_input_test.py" "${OCTO_GODOT_EXECUTABLE}" $<TARGET_FILE:octo_server> "${CMAKE_CURRENT_SOURCE_DIR}/game" "${CMAKE_CURRENT_BINARY_DIR}")
   set_tests_properties(game.input PROPERTIES TIMEOUT 45)
  endif()
 endif()
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/benchmark.cpp")
 add_executable(octo_benchmark tools/benchmark.cpp)
 target_link_libraries(octo_benchmark PRIVATE octo_physics)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_tests.cpp")
 add_executable(octo_integration_tests tests/integration_tests.cpp)
 target_link_libraries(octo_integration_tests PRIVATE octo_physics)
 add_test(NAME physics.integration COMMAND octo_integration_tests)
endif()
