cmake_minimum_required(VERSION 3.25)
project(octo_rebuild VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OCTO_ASAN "Enable AddressSanitizer" OFF)
add_library(octo_physics core/world.cpp)
target_include_directories(octo_physics PUBLIC include)
if(MSVC)
 target_compile_options(octo_physics PRIVATE /W4 /permissive- /utf-8 /fp:precise)
 if(OCTO_ASAN)
  target_compile_options(octo_physics PUBLIC /fsanitize=address /Z7)
  target_link_options(octo_physics PUBLIC /INCREMENTAL:NO)
 endif()
else()
 target_compile_options(octo_physics PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/physics_tests.cpp")
 add_executable(octo_physics_tests tests/physics_tests.cpp)
 target_link_libraries(octo_physics_tests PRIVATE octo_physics)
 add_test(NAME physics.unit COMMAND octo_physics_tests)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/bridge/server.cpp")
 add_executable(octo_server bridge/server.cpp)
 target_link_libraries(octo_server PRIVATE octo_physics)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/benchmark.cpp")
 add_executable(octo_benchmark tools/benchmark.cpp)
 target_link_libraries(octo_benchmark PRIVATE octo_physics)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_tests.cpp")
 add_executable(octo_integration_tests tests/integration_tests.cpp)
 target_link_libraries(octo_integration_tests PRIVATE octo_physics)
 add_test(NAME physics.integration COMMAND octo_integration_tests)
endif()
